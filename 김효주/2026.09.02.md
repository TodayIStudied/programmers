### 접근 방식 : ''.join(sorted(s, reverse=True))
- 왜 이 방식으로 풀었나?
'''
def solution(s):
    ans = sorted(s)
    small = []
    big = []
    total = ''
    
    for idx, char in enumerate(ans):
        if char == ans[idx].lower():
            small.append(char)
        else:
            big.append(char)
            
    small = sorted(small, reverse=True)
    big = sorted(big, reverse=True)
    total = ''.join(small + big)
    return total
'''

sorted가 대문자는 구분못하는 줄 알고 나눠서 더하는 걸로 했는데, 다른 사람이 한 걸 보니 기가 차서 말이 안나온다.

return ''.join(sorted(s, reverse=True))
